PROGRAM 5
STACK STORAGE ALLOCATION STRATEGY


#include <stdio.h>

int main()
{
    int stack[5], top = -1, n, i;

    printf("Enter number of elements (max 5): ");
    scanf("%d", &n);

    for(i = 0; i < n; i++)
    {
        scanf("%d", &stack[++top]);
    }

    printf("Stack elements:\n");

    while(top != -1)
        printf("%d\n", stack[top--]);

    return 0;
}

PROGRAM 6
RECURSIVE DESCENT PARSER

GRAMMAR:
E -> E + T | T
T -> T * F | F
F -> ( E ) | id


#include <stdio.h>
char s[50];
int i = 0;

int F()
{
    if(s[i] == 'i') { i++; return 1; }
    return 0;
}

int T()
{
    if(!F()) return 0;
    while(s[i] == '*')
    {
        i++;
        if(!F()) return 0;
    }
    return 1;
}

int E()
{
    if(!T()) return 0;
    while(s[i] == '+')
    {
        i++;
        if(!T()) return 0;
    }
    return 1;
}

int main()
{
    printf("Enter expression: ");
    scanf("%49s", s);

    if(E() && s[i] == '\0')
        printf("Valid");
    else
        printf("Invalid");

    return 0;
}

PROGRAM 7
RECURSIVE DESCENT PARSER

GRAMMAR:
S -> L
L -> L , S | S
S -> a

#include <stdio.h>
char s[50];
int i = 0;

int S();
int L();

int S()
{
    if(s[i] == 'a')
    {
        i++;
        return 1;
    }
    if(s[i] == '(')
    {
        i++;
        if(L() && s[i] == ')')
        {
            i++;
            return 1;
        }
    }
    return 0;
}

int L()
{
    if(!S()) return 0;
    while(s[i] == ',')
    {
        i++;
        if(!S()) return 0;
    }
    return 1;
}

int main()
{
    printf("Enter string: ");
    scanf("%49s", s);

    if(S() && s[i] == '\0')
        printf("Valid");
    else
        printf("Invalid");

    return 0;
}

PROGRAM 18
LEX PROGRAM TO IMPLEMENT ARITHMETIC OPERATIONS


%{
#include <stdio.h>
%}

%%
[0-9]+[+\-*/][0-9]+  {
    int a, b;
    char op;
    sscanf(yytext, "%d%c%d", &a, &op, &b);

    if(op == '+') printf("Result = %d\n", a+b);
    if(op == '-') printf("Result = %d\n", a-b);
    if(op == '*') printf("Result = %d\n", a*b);
    if(op == '/' && b != 0) printf("Result = %d\n", a/b);
    if(op == '/' && b == 0) printf("Cannot divide by zero\n");
}
[ \t\n]+ ;
.       printf("Invalid expression\n");
%%

int main()
{
    printf("Enter expression: ");
    yylex();
    return 0;
}
